#include <stdio.h>
#include <string.h>

// Function to get first day of the year
int getFirstDayOfTheYear(int year)
{
    int day = (year * 365 + ((year - 1) / 4) - ((year - 1) / 100) + ((year - 1) / 400)) % 7;
    return day;
}

// Function to calculate age
void birthdayAgeCalculator() {
    struct Date {
        int day, month, year;
    } birth, today;

    int ageYear, ageMonth, ageDay;

    printf("\n===== HAPPY BIRTHDAY AGE CALCULATOR =====\n");
    printf("Enter your date of birth (dd mm yyyy): ");
    scanf("%d %d %d", &birth.day, &birth.month, &birth.year);

    printf("Enter today's date (dd mm yyyy): ");
    scanf("%d %d %d", &today.day, &today.month, &today.year);

    // Adjust for borrowing
    if (today.day < birth.day) {
        today.day += 30;
        today.month -= 1;
    }

    if (today.month < birth.month) {
        today.month += 12;
        today.year -= 1;
    }

    // Calculate age
    ageDay = today.day - birth.day;
    ageMonth = today.month - birth.month;
    ageYear = today.year - birth.year;

    printf("\nYou are %d years, %d months and %d days old.\n", ageYear, ageMonth, ageDay);

    if (ageMonth == 0 && ageDay == 0)
        printf("🎉🎂 HAPPY BIRTHDAY! 🎂🎉\n");
}

// Function to show holidays
void showHolidays() {
    int choice, month;
    printf("\nDo you want to see holidays? (1 = Yes, 0 = No): ");
    scanf("%d", &choice);

    if (choice == 1) {
        printf("\nSee holidays for (1 = Whole Year, 2 = Single Month): ");
        scanf("%d", &month);

        if (month == 1) {
            printf("\n========= 🇧🇩 BANGLADESH NATIONAL HOLIDAYS =========\n");
            printf("শহীদ দিবস ও আন্তর্জাতিক মাতৃভাষা দিবস - ২১ ফেব্রুয়ারি\n");
            printf("স্বাধীনতা ও জাতীয় দিবস - ২৬ মার্চ\n");
            printf("বাংলা নববর্ষ - ১৪ এপ্রিল\n");
            printf("বিজয় দিবস - ১৬ ডিসেম্বর\n");
        } 
        else if (month == 2) {
            int m;
            printf("\nEnter month number (1-12): ");
            scanf("%d", &m);

            switch (m) {
                case 2:
                    printf("\nশহীদ দিবস ও আন্তর্জাতিক মাতৃভাষা দিবস - ২১ ফেব্রুয়ারি\n");
                    break;
                case 3:
                    printf("\nস্বাধীনতা ও জাতীয় দিবস - ২৬ মার্চ\n");
                    break;
                case 4:
                    printf("\nবাংলা নববর্ষ - ১৪ এপ্রিল\n");
                    break;
                case 12:
                    printf("\nবিজয় দিবস - ১৬ ডিসেম্বর\n");
                    break;
                default:
                    printf("\nNo national holiday this month.\n");
            }
        } 
        else {
            printf("\nInvalid choice.\n");
        }
    } 
    else {
        printf("\nOkay, have a great day! 😊\n");
    }
}

int main() {
    char *months[] =  {" JANUARY", "FEBRUARY", "MARCH", "APRIL", "MAY", "JUNE",
                       "JULY", "AUGUST", "SEPTEMBER", "OCTOBER", "NOVEMBER", "DECEMBER"};
    int daysInMonth[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    int i, j, totalDays, weekDay = 0, spaceCounter = 0, year;

    printf("Enter your favorite year : ");
    scanf("%d", &year);

    printf("\n\n\n*************** WELCOME TO %d *********************\n\n", year);

    if ((year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)) {
        daysInMonth[1] = 29;
    }

    weekDay = getFirstDayOfTheYear(year);

    for (i = 0; i < 12; i++) {
        printf("\n\n\n--------------------%s--------------------------\n", months[i]);
        printf("\n    SUN    MON    TUE    WED    THU    FRI    SAT\n\n");

        for (spaceCounter = 1; spaceCounter <= weekDay; spaceCounter++) {
            printf("       ");
        }

        totalDays = daysInMonth[i];
        for (j = 1; j <= totalDays; j++) {
            printf("%7d", j);
            weekDay++;
            if (weekDay > 6) {
                weekDay = 0;
                printf("\n");
            }
        }
    }

    // Ask if user wants to know their age
    int choice;
    printf("\n\nDo you want to know your age? (1 = Yes, 0 = No): ");
    scanf("%d", &choice);

    if (choice == 1) {
        birthdayAgeCalculator();
    } else {
        printf("\nOkay, skipping age calculation.\n");
    }

    // After that, ask about holidays
    showHolidays();

    return 0;
}
